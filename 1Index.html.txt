        // ==========================================
        // 3. ӨЗГЕРМАЛЫЛАР ЖӘНЕ КҮЙ (STATE)
        // ==========================================
        let studentData = { name: '', classVal: '', variant: 1 };
        let currentQuestions = [];
        let userAnswers = new Array(20).fill(null);
        let currentQuestionIndex = 0;
        let timerInterval = null;
        let timeLeft = 30 * 60; // 30 минут секундпен

        // ==========================================
        // 4. ТЕСТТІ БАСТАУ
        // ==========================================
        function startQuiz(event) {
            event.preventDefault();
            
            const nameInput = document.getElementById('studentName').value.trim();
            const classInput = document.getElementById('studentClass').value.trim();
            const variantInput = parseInt(document.getElementById('variantSelect').value);

            if (!nameInput || !classInput) return;

            studentData.name = nameInput;
            studentData.classVal = classInput;
            studentData.variant = variantInput;

            currentQuestions = quizData[variantInput];

            // UI жаңарту
            document.getElementById('startModal').classList.add('hidden');
            document.getElementById('quizApp').classList.remove('hidden');

            document.getElementById('displayStudentInfo').innerText = `${studentData.name} (${studentData.classVal})`;
            document.getElementById('displaySubjectInfo').innerText = `Информатика • ${studentData.variant}-нұсқа`;

            initQuestionGrid();
            renderQuestion();
            startTimer();
        }

        // ==========================================
        // 5. ТАЙМЕР ЛОГИКАСЫ (30 МИНУТ)
        // ==========================================
        function startTimer() {
            updateTimerDisplay();
            timerInterval = setInterval(() => {
                timeLeft--;
                updateTimerDisplay();

                if (timeLeft <= 0) {
                    clearInterval(timerInterval);
                    alert("Уақытыңыз аяқталды! Тест автоматты түрде тапсырылады.");
                    finishQuiz();
                }
            }, 1000);
        }

        function updateTimerDisplay() {
            const minutes = Math.floor(timeLeft / 60);
            const seconds = timeLeft % 60;
            const formatted = `${minutes.toString().padStart(2, '0')}:${seconds.toString().padStart(2, '0')}`;
            const timerEl = document.getElementById('timer');
            timerEl.innerText = formatted;

            if (timeLeft < 300) { // 5 минуттан аз қалса
                document.getElementById('timerContainer').classList.remove('text-amber-400');
                document.getElementById('timerContainer').classList.add('text-rose-500', 'border-rose-500/50', 'bg-rose-950/20');
            }
        }

        // ==========================================
        // 6. СҰРАҚТАР МЕН НАВИГАЦИЯ
        // ==========================================
        function initQuestionGrid() {
            const grid = document.getElementById('questionGrid');
            grid.innerHTML = '';
            for (let i = 0; i < 20; i++) {
                const btn = document.createElement('button');
                btn.id = `grid-btn-${i}`;
                btn.innerText = i + 1;
                btn.onclick = () => jumpToQuestion(i);
                btn.className = `h-9 w-full rounded-lg font-semibold text-xs transition border flex items-center justify-center bg-slate-900 border-slate-700 text-slate-400 hover:border-indigo-500`;
                grid.appendChild(btn);
            }
        }

        function renderQuestion() {
            const q = currentQuestions[currentQuestionIndex];
            
            // Бас тақырып және белгі
            document.getElementById('questionBadge').innerText = `Сұрақ ${currentQuestionIndex + 1} / 20`;
            document.getElementById('questionText').innerText = q.q;

            // Жауаптар
            const optionsContainer = document.getElementById('optionsContainer');
            optionsContainer.innerHTML = '';

            const optionLetters = ['A', 'B', 'C', 'D'];

            q.options.forEach((opt, idx) => {
                const isSelected = userAnswers[currentQuestionIndex] === idx;
                const btn = document.createElement('button');
                btn.onclick = () => selectOption(idx);
                btn.className = `w-full text-left p-4 rounded-xl border transition flex items-center space-x-3 ${
                    isSelected 
                        ? 'bg-indigo-600/20 border-indigo-500 text-white shadow-lg shadow-indigo-500/10' 
                        : 'bg-slate-900/60 border-slate-700/80 text-slate-300 hover:bg-slate-700/50 hover:border-slate-600'
                }`;

                btn.innerHTML = `
                    <span class="w-8 h-8 rounded-lg flex items-center justify-center font-bold text-xs ${
                        isSelected ? 'bg-indigo-500 text-white' : 'bg-slate-800 text-slate-400 border border-slate-700'
                    }">${optionLetters[idx]}</span>
                    <span class="text-sm font-medium flex-1">${opt}</span>
                `;
                optionsContainer.appendChild(btn);
            });

            // Навигация түймелерінің күйі
            document.getElementById('prevBtn').disabled = currentQuestionIndex === 0;
            const nextBtn = document.getElementById('nextBtn');
            if (currentQuestionIndex === 19) {
                nextBtn.innerHTML = `Соңғы сұрақ <i class="fa-solid fa-flag-checkered ml-1"></i>`;
            } else {
                nextBtn.innerHTML = `Келесі <i class="fa-solid fa-chevron-right ml-1"></i>`;
            }

            updateProgress();
            updateGridStyle();
        }

        function selectOption(index) {
            userAnswers[currentQuestionIndex] = index;
            renderQuestion();
        }

        function prevQuestion() {
            if (currentQuestionIndex > 0) {
                currentQuestionIndex--;
                renderQuestion();
            }
        }

        function nextQuestion() {
            if (currentQuestionIndex < 19) {
                currentQuestionIndex++;
                renderQuestion();
            }
        }

        function jumpToQuestion(index) {
            currentQuestionIndex = index;
            renderQuestion();
        }

        function updateProgress() {
            const answeredCount = userAnswers.filter(a => a !== null).length;
            document.getElementById('progressText').innerText = `${answeredCount} / 20`;
            const percentage = (answeredCount / 20) * 100;
            document.getElementById('progressBar').style.width = `${percentage}%`;
        }

        function updateGridStyle() {
            for (let i = 0; i < 20; i++) {
                const btn = document.getElementById(`grid-btn-${i}`);
                const isCurrent = i === currentQuestionIndex;
                const isAnswered = userAnswers[i] !== null;

                btn.className = `h-9 w-full rounded-lg font-semibold text-xs transition border flex items-center justify-center `;

                if (isCurrent) {
                    btn.className += `ring-2 ring-indigo-400 border-transparent bg-indigo-600 text-white font-bold`;
                } else if (isAnswered) {
                    btn.className += `bg-emerald-500/10 border-emerald-500/40 text-emerald-400`;
                } else {
                    btn.className += `bg-slate-900 border-slate-700 text-slate-400 hover:border-slate-500`;
                }
            }
        }

        // 7. ТЕСТТІ АЯҚТАУ ЖӘНЕ НӘТИЖЕНІ ЖІБЕРУ
        // ==========================================
        function confirmSubmission() {
            document.getElementById('confirmModal').classList.remove('hidden');
        }

        function closeConfirmModal() {
            document.getElementById('confirmModal').classList.add('hidden');
        }

        function finishQuiz() {
            clearInterval(timerInterval);
            closeConfirmModal();

            // Баллды есептеу
            let score = 0;
            currentQuestions.forEach((q, idx) => {
                if (userAnswers[idx] === q.correct) {
                    score++;
                }
            });

            const total = 20;
            const percentageVal = Math.round((score / total) * 100);
            const percentageStr = `${percentageVal}%`;

            // Уақытты пішімдеу (күн мен уақыт)
            const now = new Date();
            const dateStr = now.getFullYear() + '-' +
                String(now.getMonth() + 1).padStart(2, '0') + '-' +
                String(now.getDate()).padStart(2, '0') + ' ' +
                String(now.getHours()).padStart(2, '0') + ':' +
                String(now.getMinutes()).padStart(2, '0');

            // JSON Пайлоад дайындау
            const payload = {
                date: dateStr,
                studentName: studentData.name,
                studentClass: studentData.classVal,
                subjectTitle: `Информатика (${studentData.variant}-нұсқа)`,
                score: score,
                total: total,
                percentage: percentageStr
            };

            // Нәтиже экранын көрсету
            document.getElementById('quizApp').classList.add('hidden');
            document.getElementById('resultModal').classList.remove('hidden');

            document.getElementById('resultStudentInfo').innerText = `${studentData.name} • ${studentData.classVal} сыныбы`;
            document.getElementById('resultScore').innerText = `${score} / ${total}`;
            document.getElementById('resultPercentage').innerText = percentageStr;

            // Google Apps Script-ке POST сұранысын жіберу
            sendResultsToGoogleScript(payload);
        }

        async function sendResultsToGoogleScript(payload) {
            const sendStatusEl = document.getElementById('sendStatus');

            try {
                // Apps Script үшін 'no-cors' немесе text/plain стандартын қолданамыз
                await fetch(GOOGLE_SCRIPT_URL, {
                    method: 'POST',
                    mode: 'no-cors',
                    headers: {
                        'Content-Type': 'application/json'
                    },
                    body: JSON.stringify(payload)
                });

                // Google Apps Script no-cors режимінде 200 қайтарады
                sendStatusEl.className = "p-3 rounded-xl bg-emerald-950/40 text-xs text-emerald-400 border border-emerald-800/40 flex items-center justify-center space-x-2";
                sendStatusEl.innerHTML = `<i class="fa-solid fa-circle-check text-base"></i><span>Нәтиже Google Sheets кестесіне сәтті сақталды! ✅</span>`;
            } catch (err) {
                console.error("Жіберу қатесі:", err);
                sendStatusEl.className = "p-3 rounded-xl bg-rose-950/40 text-xs text-rose-400 border border-rose-800/40 flex items-center justify-center space-x-2";
                sendStatusEl.innerHTML = `<i class="fa-solid fa-triangle-exclamation text-base"></i><span>Серверге жіберу кезінде қате орын алды.</span>`;
            }
        }
    </script>
